#   （且 rrpg_plus 的 scipy 1.9.3 本就要求 numpy<1.26.0，已处于
#    ABI 边缘状态，任何扰动都可能触发难以定位的数值错误）。
#
#   隔离已验证：安装本环境后 rrpg_plus 的 90 个包保持逐字节一致，
#   基线 md5 058da4938e6f1b161c4fad5bce0161bf。
#
# 重建方式：
#   conda create -n audio_linux python=3.11 -y
#   conda activate audio_linux
#   pip install -r requirements.txt
#   # GPU 版 torch 需按下方注释单独装
